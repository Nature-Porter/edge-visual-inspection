cmake_minimum_required(VERSION 3.24)
project(edge_visual_inspection VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(INSPECTION_CUDA "Build CUDA backend and kernels" ON)
option(INSPECTION_TRT "Build TensorRT runtime backend" ON)
option(INSPECTION_QT_UI "Build optional Qt Widgets desktop frontend" OFF)
set(ORT_ROOT "" CACHE PATH "Extracted onnxruntime-win-x64-gpu-1.22.0")
set(TRT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/TensorRT-10.9.0.34" CACHE PATH "TensorRT 10.9 SDK")
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann/include/nlohmann/json.hpp")
  add_library(inspection_json INTERFACE)
  target_include_directories(inspection_json INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann/include")
  add_library(nlohmann_json::nlohmann_json ALIAS inspection_json)
else()
  message(FATAL_ERROR "Local nlohmann/json v3.11.3 header missing. User-managed downloads only: place json.hpp in third_party/nlohmann/include/nlohmann/.")
endif()
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ORT_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${ORT_ROOT}/lib" REQUIRED)
add_library(inspection_core src/core/contracts.cpp src/backends/ort_backend.cpp)
target_include_directories(inspection_core PUBLIC include ${ORT_INCLUDE} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(inspection_core PUBLIC nlohmann_json::nlohmann_json ${ORT_LIBRARY} ${OpenCV_LIBS})
if(WIN32)
  target_link_libraries(inspection_core PUBLIC bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(inspection_core PUBLIC OpenSSL::Crypto)
endif()
if(INSPECTION_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit 12 REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  add_library(inspection_cuda src/cuda/postprocess.cu)
  set_target_properties(inspection_cuda PROPERTIES CUDA_ARCHITECTURES 89)
  target_include_directories(inspection_cuda PUBLIC include)
  target_link_libraries(inspection_cuda PUBLIC CUDA::cudart)
  target_compile_definitions(inspection_core PUBLIC INSPECTION_CUDA=1)
  target_link_libraries(inspection_core PUBLIC inspection_cuda CUDA::cudart)
endif()
if(MSVC)
  target_compile_options(inspection_core PRIVATE /W4 /utf-8)
endif()
add_executable(inspection src/app/main.cpp)
target_link_libraries(inspection PRIVATE inspection_core)
if(INSPECTION_QT_UI)
  find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets)
  add_executable(inspection_gui WIN32 src/gui/main.cpp)
  target_link_libraries(inspection_gui PRIVATE Qt5::Widgets)
  if(MSVC)
    target_compile_options(inspection_gui PRIVATE /W4 /utf-8)
  endif()
endif()
if(INSPECTION_TRT AND INSPECTION_CUDA)
  find_path(TRT_INCLUDE NvInfer.h HINTS "${TRT_ROOT}/include" REQUIRED)
  find_library(TRT_LIBRARY nvinfer_10 HINTS "${TRT_ROOT}/lib" REQUIRED)
  target_sources(inspection PRIVATE src/backends/trt_backend.cpp)
  target_include_directories(inspection PRIVATE ${TRT_INCLUDE})
  target_link_libraries(inspection PRIVATE ${TRT_LIBRARY})
  target_compile_definitions(inspection PRIVATE INSPECTION_TRT=1)
endif()
enable_testing()
add_executable(core_tests tests/unit/core_tests.cpp)
target_link_libraries(core_tests PRIVATE inspection_core)
add_test(NAME core COMMAND core_tests)
if(INSPECTION_CUDA)
  add_executable(cuda_tests tests/unit/cuda_tests.cpp)
  target_link_libraries(cuda_tests PRIVATE inspection_core)
  add_test(NAME cuda_reference COMMAND cuda_tests)
endif()
file(GLOB ORT_DLLS "${ORT_ROOT}/lib/*.dll")
foreach(dll ${ORT_DLLS})
  add_custom_command(TARGET inspection POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:inspection>")
endforeach()
